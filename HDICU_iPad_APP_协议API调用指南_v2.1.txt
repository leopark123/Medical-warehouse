[3] ⚠ 重要提醒(先看这里)
[4] 本协议的 0x03 写参数是【整体状态写入】模型,不是【单字段增量更新】。
[6] 不能发送形如 “只设置要改的那个字节,其余置 0” 的帧!!!
[8] 错误示例(会被拒绝 + 关闭其他正在运行的功能):
[9] AA 03 1E 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 01 ... CC 55
         ↑                                          ↑
         其他全 0                             只设 light_ctrl = 0x01

后果:
  ① nursing_level = 0  → OOB 拒绝,返回 0x04 带 IPAD_ERR_NURSING_OOB (0x08)
                         整帧作废,light 也不会被改!
  ② 即便通过,其他字段 0 会:
     - fog_time = 0     → 停止正在运行的雾化倒计时
     - disinfect_time=0 → 停止消毒倒计时
     - inner_cycle = 0  → 关闭内循环
     - fresh_air   = 0  → 关闭新风
     - open_o2     = 0  → 关闭开放供氧
[11] 正确做法: APP 本地缓存完整状态,每次发 0x03 时把所有字段回填当前值,只改变要改的那一位。
[13] 1. 协议概览
[14] 1.1 物理层
[15] 1.2 帧格式
[16] 发送 (APP → 主控):  AA [CMD] [LEN] [DATA...] [CS] 55
应答 (主控 → APP):  AA [CMD] [LEN] [DATA...] [CS] 55

- 帧头:   固定 0xAA (1 字节)
- 命令码: CMD (1 字节)
- 长度:   LEN (1 字节) = DATA 部分字节数
- 数据:   DATA (LEN 字节)
- 校验:   CS (1 字节) = (0xAA + CMD + LEN + 全部 DATA 字节) & 0xFF
          ⚠ 校验和 INCLUDES 帧头 0xAA (屏幕协议不同,不要混淆)
- 帧尾:   固定 0x55 (1 字节)
[17] 1.3 字节序
[18] 所有多字节数值均为【大端】(Big-Endian,高字节先)
[19] 例: target_temp = 250 (25.0℃ x10)  → 字节序: 0x00 0xFA
     target_humidity = 500 (50.0%)   → 字节序: 0x01 0xF4
     target_o2 = 210 (21.0%)         → 字节序: 0x00 0xD2
     target_co2 = 1000 (ppm)         → 字节序: 0x03 0xE8

[24] 2.1 APP 启动流程
[25] ┌─────────────────────────────────────────────────┐
│ 1. 建立串口连接 (115200/8N1)                      │
├─────────────────────────────────────────────────┤
│ 2. 发 0x01 查询 (LEN=0)                           │
│    → 等待 0x02 回复 (46 字节)                     │
│    → 把所有字段解析到 APP 本地缓存                │
├─────────────────────────────────────────────────┤
│ 3. 发 0x0A 读限值 (LEN=0)                         │
│    → 等待 0x0A 回复 (18 字节)                     │
│    → 保存到本地,后续 UI 限值用                    │
├─────────────────────────────────────────────────┤
│ 4. 启动 500ms 轮询:                               │
│    - 每 500ms 发 0x05 → 获取实时数据 0x06 (20B)    │
│    - 实时数据用于更新 UI(温湿 O2 / 告警/ 定时)   │
├─────────────────────────────────────────────────┤
│ 5. 用户在 UI 操作时:                              │
│    a. 更新 APP 本地缓存                           │
│    b. 整帧发送 0x03 (30 字节,含所有字段)          │
│    c. 等待 0x04 ACK → 检查是否 OOB                │
│    d. 如果 OOB → 回滚本地缓存 + 提示用户          │
└─────────────────────────────────────────────────┘

[43] 4.2 构建 0x03 写入帧
[44] /* 辅助: 大端写入 u16 */
static void put_u16_be(uint8_t *dst, uint16_t v) {
    dst[0] = (uint8_t)(v >> 8);
    dst[1] = (uint8_t)(v & 0xFF);
}

/* 构建 0x03 帧, 返回总长度 (固定 35 字节) */
uint16_t build_write_params_frame(uint8_t *frame_out) {
    uint8_t data[30];
    memset(data, 0, 30);

    put_u16_be(&data[0],  g_local_cache.target_temp);
    put_u16_be(&data[2],  g_local_cache.target_humidity);
    put_u16_be(&data[4],  g_local_cache.target_o2);
    put_u16_be(&data[6],  g_local_cache.target_co2);
    put_u16_be(&data[8],  g_local_cache.fog_time);
    put_u16_be(&data[10], g_local_cache.disinfect_time);
    data[12] = g_local_cache.fan_speed;
    data[13] = g_local_cache.nursing_level;   /* !! 不能是 0 */
    data[14] = g_local_cache.inner_cycle;
    data[15] = g_local_cache.fresh_air;
    data[16] = g_local_cache.open_o2;
    data[17] = g_local_cache.light_ctrl;
    data[18] = 1;   /* cancel_flags: 1 = 接受更新 */
    data[19] = 1;
    data[20] = 1;
    data[21] = 1;
    put_u16_be(&data[22], (uint16_t)g_local_cache.cal_temp);
    put_u16_be(&data[24], (uint16_t)g_local_cache.cal_humid);
    put_u16_be(&data[26], (uint16_t)g_local_cache.cal_o2);
    put_u16_be(&data[28], (uint16_t)g_local_cache.cal_co2);

    /* 帧头 + 命令 + 长度 */
    frame_out[0] = 0xAA;
    frame_out[1] = 0x03;
    frame_out[2] = 30;
    memcpy(&frame_out[3], data, 30);

    /* 校验: AA + CMD + LEN + 所有 DATA */
    uint32_t cs = 0xAA + 0x03 + 30;
    for (int i = 0; i < 30; i++) cs += data[i];
    frame_out[33] = (uint8_t)(cs & 0xFF);
    frame_out[34] = 0x55;

    return 35;
}

void on_user_tap_inspection_light(void) {
    g_local_cache.light_ctrl ^= 0x01;   /* 翻转 bit0 */

    uint8_t frame[35];

/* 用户通过 UI 滚动条调节温度目标 */
void on_user_set_target_temp(float temp_celsius) {
    uint16_t new_temp = (uint16_t)(temp_celsius * 10.0f + 0.5f);

    /* 预检查:APP 侧也做一次 OOB, 避免来回 */
    if (new_temp < 100 || new_temp > 400) {
        show_ui_error("温度设定需在 10.0-40.0℃ 之间");
        return;
    }

    g_local_cache.target_temp = new_temp;

    uint8_t frame[35];
    uint16_t len = build_write_params_frame(frame);
    uart_send(frame, len);
}
[47] 4.4 处理 0x04 ACK
[48] typedef enum {
    ACK_OK = 0,
    ACK_OOB,
    ACK_CMD_ERR,
    ACK_TIMEOUT
} AckResult_t;

AckResult_t handle_0x04_ack(const uint8_t *data, uint8_t len) {
    if (len != 2) return ACK_CMD_ERR;

    uint8_t result = data[0];
    uint8_t err    = data[1];

    if (result == 0x00) {
        return ACK_OK;
    } else if (result == 0x01) {
        /* OOB, 根据 err 给用户提示 */
        const char *msg;
        switch (err) {
        case 0x01: msg = "温度超出范围"; break;
        case 0x02: msg = "湿度超出范围"; break;
        case 0x03: msg = "O2 超出范围"; break;
        case 0x04: msg = "CO2 超出范围"; break;
        case 0x05: msg = "雾化时长超限"; break;
        case 0x06: msg = "消毒时长超限"; break;
        case 0x07: msg = "风速值无效(必须 0-3)"; break;
        case 0x08: msg = "护理等级等开关参数无效"; break;
        case 0x09: msg = "校准值超范围"; break;
        case 0x0A: msg = "出厂限值无效"; break;
        default:   msg = "未知错误"; break;
        }
        show_ui_error(msg);
        /* 回滚本地缓存 */
        resync_from_device();
        return ACK_OOB;
    }
    return ACK_CMD_ERR;
}
[49] 4.5 解析 0x02 响应(读参数)
[50] /* 0x02 回复 46 字节, 用于启动时同步本地缓存 */
void parse_0x02_params(const uint8_t *data, uint8_t len) {
    if (len != 46) return;

    g_local_cache.target_temp     = ((uint16_t)data[0] << 8) | data[1];
    g_local_cache.target_humidity = ((uint16_t)data[2] << 8) | data[3];
    g_local_cache.target_o2       = ((uint16_t)data[4] << 8) | data[5];
    g_local_cache.target_co2      = ((uint16_t)data[6] << 8) | data[7];
    g_local_cache.fog_time        = ((uint16_t)data[8] << 8) | data[9];
    g_local_cache.disinfect_time  = ((uint16_t)data[10] << 8) | data[11];
    g_local_cache.fan_speed       = data[12];
    g_local_cache.nursing_level   = data[13];
    g_local_cache.inner_cycle     = data[14];
    g_local_cache.fresh_air       = data[15];
    g_local_cache.open_o2         = data[16];
    g_local_cache.light_ctrl      = data[17];
    /* data[18-33] 其他预留字段跳过 */
    /* data[34-37] cancel_flags (只读,APP 一般每次发都用 1) */
    /* data[38-45] calibration */
    g_local_cache.cal_temp  = (int16_t)(((uint16_t)data[38] << 8) | data[39]);
    g_local_cache.cal_humid = (int16_t)(((uint16_t)data[40] << 8) | data[41]);
    g_local_cache.cal_o2    = (int16_t)(((uint16_t)data[42] << 8) | data[43]);
    g_local_cache.cal_co2   = (int16_t)(((uint16_t)data[44] << 8) | data[45]);
}

[55] 5.2 协议管理器
[56] class CabinProtocol {
    static let shared = CabinProtocol()

    // APP 本地缓存的当前状态
    var cache = CabinParams()

    // 串口实例(这里假设你已有 SerialPort 抽象)
    var uart: SerialPort

    private init() {
        self.uart = SerialPort.shared
    }

    // MARK: - 大端辅助

    private static func putU16BE(_ data: inout [UInt8], _ offset: Int, _ value: UInt16) {
        data[offset]     = UInt8(value >> 8)
        data[offset + 1] = UInt8(value & 0xFF)
    }

    private static func putI16BE(_ data: inout [UInt8], _ offset: Int, _ value: Int16) {
        putU16BE(&data, offset, UInt16(bitPattern: value))
    }

    private static func getU16BE(_ data: [UInt8], _ offset: Int) -> UInt16 {
        return (UInt16(data[offset]) << 8) | UInt16(data[offset + 1])
    }

    // MARK: - 帧构建

    /// 构建 0x03 写入帧 (35 字节)
    func buildWriteParamsFrame() -> [UInt8] {
        var data = [UInt8](repeating: 0, count: 30)

        Self.putU16BE(&data, 0,  cache.targetTemp)
        Self.putU16BE(&data, 2,  cache.targetHumidity)
        Self.putU16BE(&data, 4,  cache.targetO2)
        Self.putU16BE(&data, 6,  cache.targetCO2)
        Self.putU16BE(&data, 8,  cache.fogTime)
        Self.putU16BE(&data, 10, cache.disinfectTime)
        data[12] = cache.fanSpeed
        data[13] = cache.nursingLevel        // !! 永远是 1-3
        data[14] = cache.innerCycle
        data[15] = cache.freshAir
        data[16] = cache.openO2
        data[17] = cache.lightCtrl
        data[18] = 1; data[19] = 1; data[20] = 1; data[21] = 1   // cancel_flags = 全接受
        Self.putI16BE(&data, 22, cache.calTemp)
        Self.putI16BE(&data, 24, cache.calHumid)
        Self.putI16BE(&data, 26, cache.calO2)
        Self.putI16BE(&data, 28, cache.calCO2)

        var frame: [UInt8] = [0xAA, 0x03, 30]
        frame.append(contentsOf: data)

        // 校验: AA + CMD + LEN + 全部 DATA
        var cs: UInt32 = 0xAA + 0x03 + 30
        for byte in data { cs += UInt32(byte) }
        frame.append(UInt8(cs & 0xFF))
        frame.append(0x55)

        return frame
    }

    // MARK: - 用户操作入口

    /// 翻转检查灯
    func toggleInspectionLight() {
        cache.lightCtrl ^= 0x01
        sendWriteAndWaitAck()
    }

    /// 设置护理等级 (1/2/3)
    func setNursingLevel(_ level: UInt8) {
        guard (1...3).contains(level) else {
            showError("护理等级必须 1-3")
            return
        }
        cache.nursingLevel = level
        sendWriteAndWaitAck()
    }

    /// 设置目标温度 (℃)
    func setTargetTemp(_ celsius: Float) {
        let x10 = UInt16((celsius * 10.0).rounded())
        guard (100...400).contains(x10) else {
            showError("温度应在 10.0-40.0℃")
            return
        }
        cache.targetTemp = x10
        sendWriteAndWaitAck()
    }

    private func sendWriteAndWaitAck() {
        let frame = buildWriteParamsFrame()
        uart.send(frame)
        // 上层用 async/await 或 Promise 等待 0x04 回复
    }
}
[57] 5.3 启动时同步缓存
[58] extension CabinProtocol {
    /// 启动时发 0x01 同步本地缓存
    func syncFromDevice() async throws {
        // 发 0x01 查询
        let cs: UInt8 = UInt8((UInt32(0xAA) + 0x01 + 0x00) & 0xFF)
        let queryFrame: [UInt8] = [0xAA, 0x01, 0x00, cs, 0x55]
        uart.send(queryFrame)

        // 等待 0x02 回复 (46 字节)
        let reply = try await waitForFrame(cmd: 0x02, expectedLen: 46, timeout: 1.0)
        parseParams(reply)
    }

    private func parseParams(_ data: [UInt8]) {
        cache.targetTemp     = Self.getU16BE(data, 0)
        cache.targetHumidity = Self.getU16BE(data, 2)
        cache.targetO2       = Self.getU16BE(data, 4)
        cache.targetCO2      = Self.getU16BE(data, 6)
        cache.fogTime        = Self.getU16BE(data, 8)
        cache.disinfectTime  = Self.getU16BE(data, 10)
        cache.fanSpeed       = data[12]
        cache.nursingLevel   = data[13]
        cache.innerCycle     = data[14]
        cache.freshAir       = data[15]
        cache.openO2         = data[16]
        cache.lightCtrl      = data[17]
        cache.calTemp        = Int16(bitPattern: Self.getU16BE(data, 38))
        cache.calHumid       = Int16(bitPattern: Self.getU16BE(data, 40))
        cache.calO2          = Int16(bitPattern: Self.getU16BE(data, 42))
        cache.calCO2         = Int16(bitPattern: Self.getU16BE(data, 44))
    }
}

        case .humidOOB:        return "湿度超出范围"
        case .o2OOB:           return "O2 超出范围"
        case .co2OOB:          return "CO2 超出范围"
        case .fogOOB:          return "雾化时长超限"
        case .disinfectOOB:    return "消毒时长超限"
        case .fanOOB:          return "风速值无效"
        case .nursingOOB:      return "护理等级等开关参数无效(如 nursing=0)"
        case .calOOB:          return "校准值超出 ±500"
        case .limitsInvalid:   return "出厂限值无效(下限 >= 上限)"
        case .cmdError:        return "命令错误"
        case .timeout:         return "设备无应答"
        case .invalidLength:   return "帧长异常"
        }
    }
}

func parseAck0x04(_ data: [UInt8]) throws {
    guard data.count == 2 else { throw CabinError.invalidLength }

    let result = data[0]
    let errCode = data[1]

    switch result {
    case 0x00: return                      // 成功
    case 0x01:
        switch errCode {
        case 0x01: throw CabinError.tempOOB
        case 0x02: throw CabinError.humidOOB
        case 0x03: throw CabinError.o2OOB
        case 0x04: throw CabinError.co2OOB
        case 0x05: throw CabinError.fogOOB
        case 0x06: throw CabinError.disinfectOOB
        case 0x07: throw CabinError.fanOOB
        case 0x08: throw CabinError.nursingOOB
        case 0x09: throw CabinError.calOOB
        case 0x0A: throw CabinError.limitsInvalid
        default:   throw CabinError.cmdError
        }
    default:
        throw CabinError.cmdError
    }
}
[62] 6. 常见错误对照
[63] 6.1 错误 #1: 只设一位其余置 0
[64] ❌ 错的:
发: AA 03 1E 00 00 00 00 00 00 00 00 00 00 00 00 00 00 01 00 00 00 ... CC 55
    (只把 data[17]=0x01 设检查灯, 其余全 0)

结果:
  - nursing_level=0 → 主控返回 0x04 带 IPAD_ERR_NURSING_OOB, 整帧作废
  - 即便能过, fog_time=0 会停雾化, disinfect_time=0 停消毒

✅ 对的:
  1. 先发 0x01 查询当前状态
  2. 收 0x02 后更新本地 cache (含 nursing=2 等当前值)
  3. cache.light_ctrl |= 0x01
  4. buildWriteParamsFrame() → 发送
  5. 等 0x04 ACK
[65] 6.2 错误 #2: 字节序搞反(小端)
[66] ❌ 错的: target_temp=250 写成 FA 00
✅ 对的: target_temp=250 写成 00 FA
       (大端, 高字节在前)
[67] 6.3 错误 #3: 校验和不含帧头
[68] ❌ 错的: cs = (cmd + len + data) & 0xFF      (屏幕协议这么算)
✅ 对的: cs = (0xAA + cmd + len + data) & 0xFF (iPad 协议包含 0xAA)

注意: iPad 和屏幕协议的校验和规则不同,不要混淆!
[69] 6.4 错误 #4: 校准值没用补码(负值)
[70] 例: cal_temp = -25 (即 -2.5℃ 校准偏移)

❌ 错的: 写成 FF 19 (负数直接取反加 1)
✅ 对的: 写成 FF E7 (补码: -25 的 int16 补码 = 0xFFE7)

Swift:  let bytes = UInt16(bitPattern: Int16(-25))  // 0xFFE7
C:      int16_t v = -25; uint16_t u = (uint16_t)v;  // 自动变 0xFFE7
[71] 6.5 错误 #5: cancel_flags 全 0 以为能“保护”温度不变
[72] ❌ 误解: “cancel_flags 全 0 的话, 温度湿度也不会被改”
✅ 真相:  cancel_flags 全 0 确实会跳过 temp/humid/o2/co2 更新
           但 fog_time / fan / nursing / cycle / fresh / open_o2 / light
           这些字段没有 cancel 机制, 还是会被写入!

           如果要“只改 light, 其他完全不动”, 必须按当前值回填所有字段。
[73] 6.6 错误 #6: 连续多次快速发 0x03
[74] 场景: 用户在 UI 上连续滑动温度滑条, APP 每 50ms 发一次 0x03

❌ 错的: 主控处理不过来, 队列溢出, 偶发丢帧

✅ 对的: APP 端做 debounce
           - 用户停止滑动 300ms 后才发
           - 或限流: 最多每 200ms 一帧
[75] 6.7 错误 #7: 忘了更新 cancel_flags 导致副作用
[76] 场景: APP 内部有 “不更新温度” 的模式 (cancel_flags[0]=0)
      用户切到另一个模式, 想恢复温度控制

❌ 错的: 只改 UI 显示, 没改 cache.sendCancelFlags
         下次发 0x03 时 cancel_flags[0] 还是 0, 温度更新被忽略

✅ 对的: 状态切换时立即同步 cancel_flags 到 cache, 并发一次完整 0x03

[79] 7.1 最小“翻转检查灯”完整帧(示例)
[80] 假设本地 cache: temp=250, humid=500, o2=210, co2=1000, fan=1, nursing=2, 其他 0, 校准 0
[81] 帧字节 (35 bytes):
  AA 03 1E                                     ← 头 + 写参 + LEN=30
  00 FA                                         ← target_temp=250
  01 F4                                         ← target_humid=500
  00 D2                                         ← target_o2=210
  03 E8                                         ← target_co2=1000
  00 00                                         ← fog_time=0
  00 00                                         ← disinf_time=0
  01                                            ← fan=1
  02                                            ← nursing=2 (关键!!!)
  00                                            ← cycle=0
  00                                            ← fresh=0
  00                                            ← open_o2=0
  01                                            ← light_ctrl=0x01 (检查灯)
  01 01 01 01                                   ← cancel_flags 全接受
  00 00 00 00 00 00 00 00                       ← 4 个校准值=0
  <CS>                                          ← 和校验
  55                                            ← 帧尾

CS 计算 (逐字节累加):
  头+命令+长度:  0xAA + 0x03 + 0x1E = 0xCB
  data[0..1]:    0x00 + 0xFA         (temp=250)
  data[2..3]:    0x01 + 0xF4         (humid=500)
  data[4..5]:    0x00 + 0xD2         (o2=210)
  data[6..7]:    0x03 + 0xE8         (co2=1000)
  data[8..11]:   0 (fog + disinf 全 0)
  data[12]:      0x01                (fan=1)
  data[13]:      0x02                (nursing=2)
  data[14..16]:  0 (cycle+fresh+open_o2)
  data[17]:      0x01                (light=0x01)
  data[18..21]:  0x01+0x01+0x01+0x01 (cancel_flags 全 1)
  data[22..29]:  0 (校准 8 字节全 0)

  累加 = 0xCB + 948 decimal = 1151 = 0x47F
  CS = 0x47F & 0xFF = 0x7F

完整帧 (35 字节):
  AA 03 1E 00 FA 01 F4 00 D2 03 E8 00 00 00 00 01 02 00 00 00 01 01 01 01 01 00 00 00 00 00 00 00 00 7F 55
